# Host checks only: this does not build a PS2 executable.
add_library(dotr_headers INTERFACE)
target_include_directories(dotr_headers INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(dotr_headers INTERFACE c_std_11)

# The decompiled C keeps the original's 32-bit words for addresses, so a 64-bit host compiler warns
# about every conversion between words and pointers; and where the original reads a local that one
# path leaves unset (e.g. Ex3d_NodeL2V's result), Clang warns too, as it does about the GCC-only
# warning names in the sources' host-build pragmas and about locals and helpers that exist only to
# reproduce the original code, a library function redeclared with the original's 32-bit prototype,
# and constants the original stores into narrow signed bit-fields. Those warnings are muted unless
# DOTR_HOST_WARNINGS is on (`make host-check WARNINGS=1`). Each flag is added only when the
# compiler knows the warning (GCC accepts any -Wno-<name> silently, so the positive form is tested).
option(DOTR_HOST_WARNINGS "Show the host warnings that only flag deliberate quirks of the decompiled C" OFF)
if(NOT DOTR_HOST_WARNINGS)
    include(CheckCCompilerFlag)
    foreach(warning int-to-pointer-cast pointer-to-int-cast int-conversion incompatible-pointer-types
                    builtin-declaration-mismatch sometimes-uninitialized
                    unknown-warning-option unused-but-set-variable unused-function
                    incompatible-library-redeclaration bitfield-constant-conversion)
        string(MAKE_C_IDENTIFIER "DOTR_HAS_W_${warning}" known)
        check_c_compiler_flag("-W${warning}" ${known})
        if(${known})
            target_compile_options(dotr_headers INTERFACE "-Wno-${warning}")
        endif()
    endforeach()
endif()
